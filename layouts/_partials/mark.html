{{- /* The Nivis mark: nested self-similar copies of one curve. h(theta) = A + B*cos(3*theta), B fixed at 1, so only the ratio A/B varies copy i is rotated by i*phi and scaled by step^i step = perfectFit(A, phi) ^ (1 - 5*fit) Each copy therefore sits inside the one before it, and rotating the series also changes how tightly it nests, because step depends on phi. Colours come from var(--mark--), generated in tokens.css from the span in data/tokens.yaml. No colour value is written here. Arguments: name, copies (optional, defaults to the mark's own), size, alt. */ -}} {{- $name := .name -}} {{- $marks := hugo.Data.marks -}} {{- $p := index $marks $name -}} {{- if not $p -}} {{- errorf "mark %q has no parameters in data/marks.yaml" $name -}} {{- end -}} {{- $a := float $p.ratio -}} {{- $copies := int (default $p.copies .copies) -}} {{- $fit := float $p.fit -}} {{- $phi := mul (float $p.rot) (div math.Pi 180.0) -}} {{- /* step > 1 turns the nesting inside out: each copy would be drawn larger than its parent. The exponent (1 - 5*fit) reaches zero at fit = 0.2, so that is the bound, and it is a property of the formula rather than a preference. */ -}} {{- if gt $fit 0.2 -}} {{- errorf "mark %q has fit %v; above 0.2 the nesting inverts because step exceeds 1" $name $fit -}} {{- end -}} {{- if lt $copies 2 -}} {{- errorf "mark %q draws %v copies; a nested series needs at least 2" $name $copies -}} {{- end -}} {{- $max := int hugo.Data.tokens.mark_span.max_copies -}} {{- if gt $copies $max -}} {{- errorf "mark %q draws %v copies, but only %v colour steps are generated" $name $copies $max -}} {{- end -}} {{- $pf := partial "mark-fit.html" (dict "a" $a "phi" $phi) -}} {{- $step := math.Pow $pf (sub 1.0 (mul 5.0 $fit)) -}} {{- $scale := div 170.0 (add $a 1.0) -}}
