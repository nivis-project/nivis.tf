{{- /* The Nivis mark: nested self-similar copies of one curve. h(theta) = A + B*cos(3*theta), B fixed at 1, so only the ratio A/B varies copy i is rotated by i*phi and scaled by step^i step = perfectFit(A, phi) ^ (1 - 5*fit) Each copy therefore sits inside the one before it, and rotating the series also changes how tightly it nests, because step depends on phi. Colours come from var(--mark-ramp-), generated in tokens.css from the span in data/tokens.yaml. A copy selects its step from its position in the series. No colour value is written here. A mark with an `animate` block in data/marks.yaml also carries that block, its resting parameters and the sampling constants as one JSON attribute, so the script that animates it holds no parameter of its own and cannot drift from what this template drew. Arguments: name, copies (optional, defaults to the mark's own), size, alt, animate (optional, emits the animation data). */ -}} {{- $points := 120 -}} {{- $fitSamples := 720 -}} {{- $name := .name -}} {{- $marks := hugo.Data.marks -}} {{- $p := index $marks $name -}} {{- if not $p -}} {{- errorf "mark %q has no parameters in data/marks.yaml" $name -}} {{- end -}} {{- $span := hugo.Data.tokens.mark_span -}} {{- $a := float $p.ratio -}} {{- $copies := int (default $p.copies .copies) -}} {{- $fit := float $p.fit -}} {{- $phi := mul (float $p.rot) (div math.Pi 180.0) -}} {{- $opacity := float $span.opacity -}} {{- $steps := int $span.ramp_steps -}} {{- /* step > 1 turns the nesting inside out: each copy would be drawn larger than its parent. The exponent (1 - 5*fit) reaches zero at fit = 0.2, so that is the bound, and it is a property of the formula rather than a preference. */ -}} {{- if gt $fit 0.2 -}} {{- errorf "mark %q has fit %v; above 0.2 the nesting inverts because step exceeds 1" $name $fit -}} {{- end -}} {{- if lt $copies 2 -}} {{- errorf "mark %q draws %v copies; a nested series needs at least 2" $name $copies -}} {{- end -}} {{- $max := int $span.max_copies -}} {{- if gt $copies $max -}} {{- errorf "mark %q draws %v copies, but only %v colour steps are generated" $name $copies $max -}} {{- end -}} {{- /* Every copy count the site supports has to land on exact ramp steps, or an animated mark and the built one disagree at rest. */ -}} {{- if ne (mod $steps (sub $copies 1)) 0 -}} {{- errorf "the colour ramp has %v steps, which %v copies does not divide evenly" $steps $copies -}} {{- end -}} {{- $data := "" -}} {{- with .animate -}} {{- $anim := $p.animate -}} {{- if not $anim -}} {{- errorf "mark %q is asked to animate but has no animate block in data/marks.yaml" $name -}} {{- end -}} {{- $r := $anim.range -}} {{- /* A range is still a set of parameters, so the formula's bounds apply to every point in it, not only to the resting pose. */ -}} {{- if gt (float (index $r.fit 1)) 0.2 -}} {{- errorf "mark %q sweeps fit up to %v; above 0.2 the nesting inverts because step exceeds 1" $name (index $r.fit 1) -}} {{- end -}} {{- if lt (int (index $r.copies 0)) 2 -}} {{- errorf "mark %q sweeps copies down to %v; a nested series needs at least 2" $name (index $r.copies 0) -}} {{- end -}} {{- if gt (int (index $r.copies 1)) $max -}} {{- errorf "mark %q sweeps copies up to %v, but only %v are supported" $name (index $r.copies 1) $max -}} {{- end -}} {{- range $k, $v := $r -}} {{- if ne (len $v) 2 -}} {{- errorf "mark %q range %q needs a low and a high value" $name $k -}} {{- end -}} {{- if ge (float (index $v 0)) (float (index $v 1)) -}} {{- errorf "mark %q range %q runs %v to %v, which is not a range" $name $k (index $v 0) (index $v 1) -}} {{- end -}} {{- end -}} {{- $rest := dict "ratio" $a "copies" $copies "rot" (float $p.rot) "fit" $fit "opacity" $opacity -}} {{- $data = dict "rest" $rest "range" $r "cycle" (float $anim.cycle_seconds) "easeIn" (float $anim.ease_in_seconds) "points" $points "fitSamples" $fitSamples "rampSteps" $steps "maxCopies" $max | jsonify -}} {{- end -}} {{- $pf := partial "mark-fit.html" (dict "a" $a "phi" $phi "samples" $fitSamples) -}} {{- $step := math.Pow $pf (sub 1.0 (mul 5.0 $fit)) -}} {{- $scale := div 170.0 (add $a 1.0) -}}
