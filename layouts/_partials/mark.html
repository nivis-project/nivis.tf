{{- /* The Nivis mark: nested self-similar copies of one curve. h(theta) = A + B*cos(k*theta), B fixed at 1, so the ratio A/B is the only control over how deep the lobes are k is the mark's own lobe count, 3 by default copy i is rotated by i*phi and scaled by step^i step = perfectFit(A, phi) ^ (1 - 5*fit) Each copy therefore sits inside the one before it, and rotating the series also changes how tightly it nests, because step depends on phi. Colours come from var(--mark-ramp-), generated in tokens.css from the span in data/tokens.yaml. A copy selects its step from its position in the series. No colour value is written here. A caller that passes an `animate` block gets it, the resting parameters and the sampling constants back as one JSON attribute, so the script that animates the mark holds no parameter of its own and cannot read a different account of where rest is than the one the build drew. Arguments: name, copies (optional, defaults to the mark's own), size, alt, animate (optional, the ranges to sweep; emits the animation data). */ -}} {{- $points := 120 -}} {{- $fitSamples := 720 -}} {{- $name := .name -}} {{- $marks := hugo.Data.marks.marks -}} {{- /* Presence is tested against the keys, not against the value. The Nivis mark states nothing, because it IS the default, so its entry is an empty map and an empty map is indistinguishable from a missing one under a truth test. */ -}} {{- $known := slice -}} {{- range $k, $_ := $marks }}{{ $known = $known | append $k }}{{ end -}} {{- if not (in $known $name) -}} {{- errorf "mark %q has no parameters in data/marks.yaml" $name -}} {{- end -}} {{- /* A mark states only where it differs from the brand default. */ -}} {{- $p := merge hugo.Data.marks.default (index $marks $name) -}} {{- $span := hugo.Data.tokens.mark_span -}} {{- $lobes := int $p.lobes -}} {{- $a := float $p.ratio -}} {{- $copies := int (default $p.copies .copies) -}} {{- $fit := float $p.fit -}} {{- $phi := mul (float $p.rot) (div math.Pi 180.0) -}} {{- $opacity := float $span.opacity -}} {{- $steps := int $span.ramp_steps -}} {{- /* step > 1 turns the nesting inside out: each copy would be drawn larger than its parent. The exponent (1 - 5*fit) reaches zero at fit = 0.2, so that is the bound, and it is a property of the formula rather than a preference. */ -}} {{- if gt $fit 0.2 -}} {{- errorf "mark %q has fit %v; above 0.2 the nesting inverts because step exceeds 1" $name $fit -}} {{- end -}} {{- if lt $copies 2 -}} {{- errorf "mark %q draws %v copies; a nested series needs at least 2" $name $copies -}} {{- end -}} {{- $max := int $span.max_copies -}} {{- if gt $copies $max -}} {{- errorf "mark %q draws %v copies, but only %v colour steps are generated" $name $copies $max -}} {{- end -}} {{- /* Every copy count the site supports has to land on exact ramp steps, or an animated mark and the built one disagree at rest. */ -}} {{- if ne (mod $steps (sub $copies 1)) 0 -}} {{- errorf "the colour ramp has %v steps, which %v copies does not divide evenly" $steps $copies -}} {{- end -}} {{- $data := "" -}} {{- with .animate -}} {{- $anim := . -}} {{- $r := $anim.range -}} {{- if not $r -}} {{- errorf "mark %q was asked to animate but the block has no range" $name -}} {{- end -}} {{- /* A range is still a set of parameters, so the formula's bounds apply to every point in it, not only to the resting pose. */ -}} {{- if gt (float (index $r.fit 1)) 0.2 -}} {{- errorf "mark %q sweeps fit up to %v; above 0.2 the nesting inverts because step exceeds 1" $name (index $r.fit 1) -}} {{- end -}} {{- if lt (int (index $r.copies 0)) 2 -}} {{- errorf "mark %q sweeps copies down to %v; a nested series needs at least 2" $name (index $r.copies 0) -}} {{- end -}} {{- if gt (int (index $r.copies 1)) $max -}} {{- errorf "mark %q sweeps copies up to %v, but only %v are supported" $name (index $r.copies 1) $max -}} {{- end -}} {{- range $k, $v := $r -}} {{- if ne (len $v) 2 -}} {{- errorf "mark %q range %q needs a low and a high value" $name $k -}} {{- end -}} {{- if ge (float (index $v 0)) (float (index $v 1)) -}} {{- errorf "mark %q range %q runs %v to %v, which is not a range" $name $k (index $v 0) (index $v 1) -}} {{- end -}} {{- end -}} {{- $rest := dict "lobes" $lobes "ratio" $a "copies" $copies "rot" (float $p.rot) "fit" $fit "opacity" $opacity -}} {{- $data = dict "rest" $rest "range" $r "cycle" (float $anim.cycle_seconds) "easeIn" (float $anim.ease_in_seconds) "points" $points "fitSamples" $fitSamples "rampSteps" $steps "maxCopies" $max | jsonify -}} {{- end -}} {{- $pf := partial "mark-fit.html" (dict "a" $a "phi" $phi "lobes" $lobes "samples" $fitSamples) -}} {{- $step := math.Pow $pf (sub 1.0 (mul 5.0 $fit)) -}} {{- /* A rotation of a whole period leaves the curve unchanged, so the exact fit is 1 and every copy is drawn at its parent's size: the series renders as one shape with its copies hidden inside it. Stated as a bound on the scale rather than as arithmetic on the rotation, because the scale is what matters and it stays true if the curve ever changes. The bound is 0.999 rather than 1 because at a whole period the two cosines are mathematically equal but are computed separately, so the ratio lands a few bits under 1 and an exact test never fires. It is a tolerance for that arithmetic and nothing more: a rotation one degree off the period gives a scale of 0.9975 and still builds, which is a nesting almost nobody can see. Where the line between tight and invisible belongs is taste, and this check does not try to settle it. */ -}} {{- $degenerate := 0.999 -}} {{- if ge $step $degenerate -}} {{- errorf "mark %q has a nesting scale of %.4f, at or above the %v bound. It rotates %v degrees between copies against a %v degree period, so every copy is drawn at its parent's size and the nesting disappears" $name $step $degenerate $p.rot (div 360.0 (float $lobes)) -}} {{- end -}} {{- $scale := div 170.0 (add $a 1.0) -}}
