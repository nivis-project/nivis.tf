{{- /* Generates a mark from data/marks.yaml at build time. Every layer is the polar curve r(t) = a + b*cos(k*t), with b = a*amp/(k^2+1), sampled at 120 points, rotated, and closed, in a -100 -100 200 200 viewBox. See section 7 of the briefing. The generator in nivis-mockup-reference.html is the reference implementation; tests/checks/mark.py reimplements the formula independently and compares every point. Arguments: name, variant (full|project|footer), size, alt (optional). Fills are custom properties so the whole family recolours in one edit. The layer sets live here, not in the data: the data answers "what shape is this mark", the variant answers "how much of it does this spot on the page show". Those change for different reasons. */ -}} {{- $name := .name -}} {{- $variant := .variant | default "full" -}} {{- $size := .size -}} {{- $alt := .alt -}} {{- $marks := hugo.Data.marks -}} {{- $p := index $marks $name -}} {{- if not $p -}} {{- errorf "mark %q has no parameters in data/marks.yaml" $name -}} {{- end -}} {{- $k := $p.k -}} {{- $amp := $p.amp -}} {{- $rot := $p.rot -}} {{- $layers := slice -}} {{- if eq $variant "full" -}} {{- $layers = slice (dict "a" 84.0 "rot" (add $rot -24.0) "fill" "mark-a" "op" "0.3") (dict "a" 80.0 "rot" (add $rot -11.0) "fill" "mark-b" "op" "0.3") (dict "a" 78.0 "rot" (add $rot 12.0) "fill" "mark-a" "op" "0.3") (dict "a" 74.0 "rot" (add $rot 25.0) "fill" "mark-b" "op" "0.3") (dict "a" 64.0 "rot" $rot "fill" "mark-core" "op" "1") -}} {{- else if eq $variant "footer" -}} {{- $layers = slice (dict "a" 80.0 "rot" (add $rot -11.0) "fill" "mark-b" "op" "0.3") (dict "a" 78.0 "rot" (add $rot 12.0) "fill" "mark-a" "op" "0.3") (dict "a" 64.0 "rot" $rot "fill" "mark-core" "op" "1") -}} {{- else if eq $variant "project" -}} {{- $layers = slice (dict "a" 84.0 "rot" (add $rot -16.0) "fill" "mark-a" "op" "0.35") (dict "a" 80.0 "rot" (add $rot 16.0) "fill" "mark-b" "op" "0.35") (dict "a" 64.0 "rot" $rot "fill" "mark-core" "op" "1") -}} {{- else -}} {{- errorf "mark variant %q is not one of full, project, footer" $variant -}} {{- end -}}
