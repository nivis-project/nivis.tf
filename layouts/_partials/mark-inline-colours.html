{{- /* Substitutes var(--mark--) with literal colours, for the standalone SVG files. A file served on its own cannot resolve custom properties. The values are computed from the span in data/tokens.yaml, the same source tokens.css generates from, so the files cannot drift from the page. No colour value originates here. Arguments: svg (the markup), copies. */ -}} {{- $out := .svg -}} {{- $n := .copies -}} {{- $m := hugo.Data.tokens.mark_span -}} {{- range $i := seq 0 (sub $n 1) -}} {{- $hue := add $m.hue_centre (mul (sub (div (float $i) (float (sub $n 1))) 0.5) $m.hue_spread) -}} {{- $c := printf "hsl(%v %v%% %v%%)" $hue $m.saturation $m.lightness -}} {{- $out = replace $out (printf "var(--mark-%v-%v)" $n $i) $c -}} {{- end -}} {{- /* safeHTML, because `replace` turns the markup into a plain string and emitting it here would escape it before the caller ever sees it. */ -}} {{- $out | safeHTML -}}
