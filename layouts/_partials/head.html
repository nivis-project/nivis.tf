{{- with .Description }} {{- end }} {{- /* Sharing metadata reads the SAME title and description the page uses. A second copy is how a preview ends up contradicting the page it previews. These tags need absolute addresses: sharing systems resolve them with no document base, so a relative og:image looks right in the markup and fails only once somebody shares the link. */ -}} {{- $canonical := .Permalink }} {{- /* A recorded theme choice must reach the document root BEFORE the first paint, which rules out an external file: the browser would paint before fetching it. So this goes inline, in the head, above the stylesheet link. The read is guarded because localStorage throws in a private window rather than returning nothing, and an unguarded throw here would stop the script and leave the page showing the wrong palette. */ -}} {{- /* One stylesheet, bundled from the sources in order. tokens.css is a template, so it is executed before it joins the bundle. Development builds skip minification and fingerprinting: a fingerprinted file and live reload fight over the cache, and the integrity attribute makes every edit a hard refresh. The gate runs against a production build. */ -}} {{- $tokensSrc := resources.Get "css/tokens.css" }} {{- if not $tokensSrc }}{{ errorf "assets/css/tokens.css is missing; the site cannot be styled without it" }}{{ end }} {{- $tokens := $tokensSrc | resources.ExecuteAsTemplate "css/tokens.css" . }} {{- $sheets := slice $tokens }} {{- range (slice "css/base.css" "css/layout.css" "css/components.css" "css/syntax.css") }} {{- $s := resources.Get . }} {{- if not $s }}{{ errorf "stylesheet %q is missing" . }}{{ end }} {{- $sheets = $sheets | append $s }} {{- end }} {{- $css := $sheets | resources.Concat "css/site.css" }} {{- if hugo.IsProduction }} {{- $css = $css | minify | fingerprint "sha256" }} {{- else }} {{- end }} {{- /* The two faces the first screen needs, before the browser finds them in the stylesheet. */}} {{- $theme := resources.Get "js/theme.js" }} {{- if not $theme }}{{ errorf "assets/js/theme.js is missing" }}{{ end }} {{- if hugo.IsProduction }}{{ $theme = $theme | minify | fingerprint "sha256" }}{{ end }}
